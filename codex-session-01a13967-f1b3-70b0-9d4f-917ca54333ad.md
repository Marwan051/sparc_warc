    def model_version(max_length: int) -> str:
        """Fingerprint model/tokenizer files and vector-affecting preprocessing."""
        model_dir = _model_directory()
        digest = hashlib.sha256(f"cls-l2-raw-text-v1|max_length={max_length}|".encode())
        required = (model_dir / "onnx" / "model_int8.onnx", model_dir / "tokenizer.json",
                    model_dir / "tokenizer_config.json", model_dir / "1_Pooling" / "config.json")
        for path in required:
            if not path.is_file():
                raise FileNotFoundError(f"embedding model asset missing: {path}")
            digest.update(path.name.encode())
            with path.open("rb") as stream:
                for block in iter(lambda: stream.read(1024 * 1024), b""):
                    digest.update(block)
        return f"bge-m3-int8-{digest.hexdigest()[:20]}"

            raise RuntimeError(f"ONNX model must expose last_hidden_state with dimension {EMBEDDING_DIMENSION}")
        if not {"input_ids", "attention_mask"}.issubset(_input_names):
            raise RuntimeError("ONNX model must accept input_ids and attention_mask")
        _output_name = output.name
    
    
    def generate_embeddings(texts: list[str]) -> np.ndarray:
        """Return L2-normalized 1024-dimensional embeddings for input texts.
    
        Set ``EMBEDDING_MODEL_DIR`` to override the default local model path.
        Batch size, token limit, and ONNX intra-op threads can be configured with
        ``EMBEDDING_BATCH_SIZE``, ``EMBEDDING_MAX_LENGTH``, and
        ``EMBEDDING_INTRA_OP_THREADS`` respectively.
        """
        if not texts:
            return np.empty((0, EMBEDDING_DIMENSION), dtype=np.float32)
        if any(not isinstance(text, str) for text in texts):
            raise TypeError("texts must contain only strings")
    
        batch_size = int(os.environ.get("EMBEDDING_BATCH_SIZE", str(DEFAULT_BATCH_SIZE)))
        max_length = int(os.environ.get("EMBEDDING_MAX_LENGTH", str(DEFAULT_MAX_LENGTH)))
        if batch_size < 1 or max_length < 1:
            raise ValueError("EMBEDDING_BATCH_SIZE and EMBEDDING_MAX_LENGTH must be positive")
    
        _load_model()
        vectors: list[np.ndarray] = []
        for start in range(0, len(texts), batch_size):
            batch = texts[start : start + batch_size]
            tokens = _tokenizer(
                batch,
                padding=True,
                truncation=True,
                max_length=max_length,
                return_tensors="np",
            )
            inputs = {
                name: np.asarray(tokens[name])
                for name in _input_names
                if name in tokens
            }
            if not inputs:
                raise RuntimeError("Tokenizer produced none of the ONNX model's required inputs")
    
            token_embeddings = _session.run([_output_name], inputs)[0]
            batch_vectors = np.asarray(token_embeddings[:, 0, :], dtype=np.float32)
            norms = np.linalg.norm(batch_vectors, axis=1, keepdims=True)
            batch_vectors /= np.maximum(norms, 1e-12)
            vectors.append(batch_vectors)
    
        result = np.concatenate(vectors, axis=0)
        if result.shape != (len(texts), EMBEDDING_DIMENSION):
            raise RuntimeError(f"Unexpected embedding shape: {result.shape}")
        return result

    def run_waves(spark, tasks, worker, commit, parallelism):
        """One partition/job per task lets completed batches commit independently.
    
        A wave has at most parallelism Spark jobs. Drain successful siblings before
        reporting failure/quota; do not cancel siblings and discard paid responses.
        """
        tasks = iter(tasks)
        completed = 0
        with ThreadPoolExecutor(max_workers=parallelism) as pool:
            while True:
                wave = list(islice(tasks, parallelism))
                if not wave:
                    return completed, False
                futures = [pool.submit(_execute, spark, worker, task) for task in wave]
                failure, stop = None, False
                committed = set()
                def save(future):
                    nonlocal completed, stop, failure
                    try:
                        result = future.result()
                        commit(result)
                        completed += 1
                        stop = stop or result.get("quota_exhausted", False)
                    except Exception as error:
                        failure = failure or error
                    committed.add(future)
                try:
                    for future in as_completed(futures):
                        save(future)
                except KeyboardInterrupt:
                    # Finish the small active wave so completed paid responses are
                    # persisted. A second interrupt still allows a forced exit.
                    for future in futures:
                        if future not in committed:
                            save(future)
                    raise
                if failure:
                    raise failure
                if stop:
                    return completed, True
    
    
    def execute_batch(spark, tasks, worker):
        """Execute an allocated batch with one partition per task, in input order.
    
        Allocation, quota accounting, and commit policy belong to the caller.
        """
        if not tasks:
            return []
        return spark.sparkContext.parallelize(tasks, len(tasks)).map(worker).collect()

        return execute_batch(spark, [task], worker)[0]

    def acquire_coordinator_lock(manifest_id):

        conn = get_connection()

    def _insert_new_records(cur, records):
        if not records:
            return []
        unique = {}
        for record in records:
            unique.setdefault(record["record_id"], record)
        records = list(unique.values())
        domains = sorted({extract_domain(r["url"]) for r in records})
        execute_values(cur, "INSERT INTO websites (domain) VALUES %s ON CONFLICT DO NOTHING", [(x,) for x in domains])
        cur.execute("SELECT domain,id FROM websites WHERE domain=ANY(%s)", (domains,))
        domain_ids = dict(cur.fetchall())
        authors = sorted({r.get("author", "").strip() for r in records
                          if r.get("author", "").strip() not in ("", "N/A")})
        author_ids = {}
        if authors:
            execute_values(cur, "INSERT INTO authors (name) VALUES %s ON CONFLICT DO NOTHING", [(x,) for x in authors])
            cur.execute("SELECT name,id FROM authors WHERE name=ANY(%s)", (authors,))
            author_ids = dict(cur.fetchall())
        rows = [(domain_ids[extract_domain(r["url"])], author_ids.get(r.get("author", "").strip()),
                 r["record_id"], r["url"], r.get("file_url"), r.get("warc_date")) for r in records]
        inserted = execute_values(
            cur,
            """INSERT INTO pages (website_id,author_id,warc_record_id,url,source_file_url,warc_date)
               VALUES %s ON CONFLICT (warc_record_id) DO NOTHING RETURNING warc_record_id,id""",
            rows, fetch=True)
        page_ids = {record_id: page_id for record_id, page_id in inserted}
        new_records = [r for r in records if r["record_id"] in page_ids]
        if new_records:
            execute_values(cur, """INSERT INTO metadata
                (page_id,title,published_date,language,arabic_dialect,word_count,char_count,links_count,headings,extraction_method)
                VALUES %s""", [(
                    page_ids[r["record_id"]], r.get("title"), r.get("published_date"), r.get("language"),
                    r.get("arabic_dialect"), r.get("word_count", 0), r.get("char_count", 0),
                    r.get("links_count", 0), json.dumps(r.get("headings", [])), r.get("extraction"),
                ) for r in new_records])
            execute_values(cur, "INSERT INTO content (page_id,cleaned_text) VALUES %s",
                           [(page_ids[r["record_id"]], r["cleaned_text"]) for r in new_records])
        return new_records

    def _insert_isolated(cur, batch, rejected):
        cur.execute("SAVEPOINT article_batch")
        try:
            for record in batch:
                parsed = urlparse(record["url"])
                if parsed.scheme not in ("http", "https") or not parsed.hostname:
                    raise ValueError("article URL must be an absolute HTTP(S) URL")
            inserted = len(_insert_new_records(cur, batch))
        except (psycopg2.DataError, psycopg2.IntegrityError, ValueError) as error:
            cur.execute("ROLLBACK TO SAVEPOINT article_batch")
            cur.execute("RELEASE SAVEPOINT article_batch")
            if len(batch) == 1:
                rejected.append({"record": batch[0], "error": str(error)})
                return 0
            return sum(_insert_isolated(cur, [record], rejected) for record in batch)
        cur.execute("RELEASE SAVEPOINT article_batch")
        return inserted

    # Spark processing stages
    
    Independent YARN jobs materialize chunks, analyze eligible chunks, and embed
    eligible chunks. They default to one concurrent Spark task per requested executor. PostgreSQL reads and writes stay
    on the driver; executors receive bounded payloads and do not need DB credentials.
    
    All four launchers (ingestion, chunking, tagging, embedding) delegate packaging and YARN
    submission to `scripts/run_processing.sh`. Spark startup and task submission live
    in `jobs/runtime.py` and `jobs/batching.py`. Ingestion retains its original Spark
    retry settings, one-partition-per-file wave, ordered commits, quotas, and recovery
    logic; chunking, tagging, and embedding use independent jobs with per-batch completion.

    ```text
    PostgreSQL articles -> driver batches -> Spark cleaning/splitting -> article_chunks
    article_chunks -> driver batches -> Spark analyzer backend -> chunk_analyses
    article_chunks -> driver batches -> Spark ONNX inference -> chunk_embeddings
    ```

    ## Batches, quotas, and recovery
    
    Defaults are 25 documents per chunking task and 10 chunks per tagging or embedding
    task, with three concurrent tasks for chunking/tagging and two for embedding by default. Input payloads are bounded to 2 MiB per task. Output
    limits are 4 MiB for chunking and 1 MiB for tagging/embedding. Override with `--input-bytes`
    and `--result-bytes`; a single oversized input or output fails explicitly. Reduce
    batch count or increase byte limits to accommodate it. A document is never
    partially materialized.
    
    Spark dynamic allocation is disabled. Ingestion uses `--max-files-per-round` for
    file concurrency. Higher task concurrency increases driver memory usage and
    simultaneous provider requests.
    
    The driver submits each task as an independent single-partition Spark job. A
    bounded wave has at most the configured number of parallel jobs, and each completed result is committed
    before waiting for slower siblings. Database inserts use bulk values. Chunk
    transactions cover one document, while tagging transactions cover one task batch.
    Selection is ordered by document/index; parallel completion order is not.
    
    Groq uses one sequential request stream per active task. A Groq task batch is not
    a provider bulk API request. Set `TAG_REQUEST_INTERVAL_SECONDS` (default 1.5) to
    pace each worker, or set `TAG_REQUESTS_PER_MINUTE` and `TAG_TOKENS_PER_MINUTE` to
    your account's shared limits. The adapter divides throughput among configured
    workers and conservatively estimates input tokens from UTF-8 bytes. This is
    rate shaping, not a cross-process quota ledger: concurrent startup requests and
    other applications sharing the account can still receive 429 responses. Workers
    honor retry delays, use at most five explicit attempts, and disable SDK retries.
    
    Daily quota exhaustion returns completed outcomes and stops later waves after
    all active siblings finish. The blocked item remains pending. Exit codes are 0
    for completion, 1 for failure, 2 for quota/long provider cooldown, and 130 for an
    interrupt. A job can finish with recorded per-item ERROR rows; inspect the printed
    counts and rerun with `--retry-errors`.
    
    Ctrl+C drains and commits the current small wave before exiting; a second
    interrupt can force an earlier exit. Each batch prints its committed counts,
    while final corpus totals are queried once at the end of the run.
    
    Speculation is disabled and Spark task retries are limited to one attempt.
    Database outcomes are idempotent, but an external call cannot be committed
    atomically with PostgreSQL. A failed task can repeat up to its batch size on
    restart; a driver crash can repeat all uncommitted active batches (up to 30 calls
    at defaults). Cumulative attempt counts cover committed outcomes, not calls lost
    to a worker/driver crash. There is no exactly-once billing guarantee.
    
    Stage-specific advisory locks prevent simultaneous coordinators of the same
    kind, while ingestion, chunking, tagging, and embedding can run independently. Changes
    committed after a job's source snapshot are picked up on the next invocation.
    
    ## Backend interface
    
    `tagging.contracts.AnalyzerBackend` accepts a list of `AnalysisRequest` and returns
    `BatchResult`, with one `AnalysisOutcome` per request. A quota stop may return a

       104            return np.empty((0, EMBEDDING_DIMENSION), dtype=np.float32)

       106            raise TypeError("texts must contain only strings")

       115        for start in range(0, len(texts), batch_size):
       116            batch = texts[start : start + batch_size]

       118                batch,
       119                padding=True,
       120                truncation=True,
       121                max_length=max_length,
       122                return_tensors="np",

        27    def run_waves(spark, tasks, worker, commit, parallelism):
        28        """One partition/job per task lets completed batches commit independently.
        29
        30        A wave has at most parallelism Spark jobs. Drain successful siblings before
        31        reporting failure/quota; do not cancel siblings and discard paid responses.
        32        """
        33        tasks = iter(tasks)
        34        completed = 0
        35        with ThreadPoolExecutor(max_workers=parallelism) as pool:
        36            while True:
        37                wave = list(islice(tasks, parallelism))
        38                if not wave:
        39                    return completed, False
        40                futures = [pool.submit(_execute, spark, worker, task) for task in wave]
        41                failure, stop = None, False
        42                committed = set()
        43                def save(future):
        44                    nonlocal completed, stop, failure
        45                    try:
        46                        result = future.result()
        47                        commit(result)
        48                        completed += 1
        49                        stop = stop or result.get("quota_exhausted", False)
        50                    except Exception as error:
        51                        failure = failure or error
        52                    committed.add(future)
        53                try:
        54                    for future in as_completed(futures):
        55                        save(future)
        56                except KeyboardInterrupt:
        57                    # Finish the small active wave so completed paid responses are
        58                    # persisted. A second interrupt still allows a forced exit.
        59                    for future in futures:
        60                        if future not in committed:
        61                            save(future)
        62                    raise
        63                if failure:
        64                    raise failure
        65                if stop:
        66                    return completed, True

        42    def commit_batch(conn, config, rows):
        43        """Commit one worker result batch; repeated matching rows are harmless."""
        44        if not rows:
        45            return
        46        keys = [(row["page_id"], config["chunking_version"], row["chunk_index"])
        47                for row in rows]
        48        if len(set(keys)) != len(keys):
        49            raise ValueError("embedding batch contains duplicate chunk identities")
        50        values = []
        51        for row in rows:
        52            vector = row["embedding"]
        53            if hashlib.sha256(row["text"].encode()).hexdigest() != row["text_hash"]:
        54                raise ValueError("embedding result text hash mismatch")
        55            if len(vector) != 1024 or any(not isinstance(x, (int, float)) for x in vector):
        56                raise ValueError("embedding vector must contain 1024 numeric values")
        57            if any(not math.isfinite(x) for x in vector):
        58                raise ValueError("embedding vector contains non-finite values")
        59            norm = math.sqrt(sum(x * x for x in vector))
        60            if not math.isclose(norm, 1.0, rel_tol=1e-3, abs_tol=1e-3):
        61                raise ValueError("embedding vector must be L2-normalized")
        62            key = keys[len(values)]
        63            values.append((*key, config["embedding_model"], config["embedding_version"],
        64                           row["text_hash"], np.asarray(vector, dtype=np.float32)))
        65        register_vector(conn)

        67            with conn.cursor() as cur:
        68                for row in rows:
        69                    cur.execute("""SELECT text_hash,eligible FROM article_chunks
        70                        WHERE page_id=%s AND chunking_version=%s AND chunk_index=%s""",

## Assistant

The key idea is:

> Spark workers only compute embeddings. The Spark driver reads PostgreSQL, divides the work, sends text to workers, receives vectors, validates them, and writes them to PostgreSQL.

The results do not go to the Spark master for database insertion. They return to the Spark **driver**. YARN/Spark’s master components mainly allocate resources and schedule executor work.

## Overall data flow

```text
PostgreSQL
    │
    │ eligible chunks without embeddings
    ▼
Spark driver on submission machine
    │
    │ batches containing text + IDs
    ▼
YARN / Spark executors
    │
    │ tokenize → ONNX inference → normalize
    ▼
Spark driver via collect()
    │
    │ validate result and commit transaction
    ▼
PostgreSQL chunk_embeddings
```

## 5. Driver initialization

## 7. How the work is distributed

This project uses a slightly unusual but intentional pattern.

For every batch, the driver runs approximately:

```python
spark.sparkContext \
    .parallelize([task], 1) \
    .map(process_batch) \
    .collect()
```

See [jobs/batching.py](/home/spark-master/rag_news_project/spark_warc_v2/jobs/batching.py:69).

Therefore:

- One embedding batch becomes one Spark job.
- That Spark job contains one partition.
- The partition is scheduled on one executor.
- `process_batch()` runs once for that partition.
- Several independent one-partition jobs run concurrently.

The driver uses a thread pool to submit a “wave” of jobs. With defaults:

```text
2 executors
2 parallel tasks
10 chunks per task
```

The approximate active layout is:

```text
Executor 1 → batch of up to 10 chunks
Executor 2 → batch of up to 10 chunks
```

When one finishes, the driver commits it. The next wave begins after all jobs in the current wave have finished.

The task itself is serialized by Spark. It contains the chunk text, identity, text hash, maximum length, and result-size limit.

## 8. Loading the model on a worker

On the executor, Spark launches:

```text
./environment/bin/python
```

That Python process imports `process_batch()` from the distributed source ZIP.

The first call eventually reaches `_load_model()` in [embeddings/embeddings.py](/home/spark-master/rag_news_project/spark_warc_v2/embeddings/embeddings.py:45).

The worker:

1. Reads `./embedding-model/1_Pooling/config.json`.
2. Confirms that the model uses CLS-token pooling rather than mean pooling.
3. Creates a sequential ONNX Runtime session.
4. Uses one ONNX intra-operation thread.
5. Loads the tokenizer locally.
6. Opens `./embedding-model/onnx/model_int8.onnx`.
7. Verifies its expected input and output interfaces.

These objects are stored in module globals:

```python
_tokenizer
_session
_input_names
_output_name
```

Spark is configured with:

```text
spark.python.worker.reuse=true
```

Consequently, the model is normally loaded once per Python worker process, not once per batch. Later tasks assigned to the same reused process use the already-open ONNX session.

If an executor or Python worker process is restarted, that new process must load the model again.

## 9. Computing the embedding

`process_batch()` extracts the text list and calls `generate_embeddings()`:

```python
vectors = generate_embeddings(
    [item["text"] for item in task["items"]]
)
```

See [embeddings/pipeline.py](/home/spark-master/rag_news_project/spark_warc_v2/embeddings/pipeline.py:13).

Inside `generate_embeddings()`:

1. Text is tokenized with padding and truncation.
2. The maximum sequence length defaults to 256.
3. The ONNX model produces `last_hidden_state`.
4. The first token vector is selected:

   ```python
   token_embeddings[:, 0, :]
   ```

   This is CLS-token pooling.

5. Each 1024-dimensional vector is converted to `float32`.
6. Each vector is L2-normalized.
7. The output shape is verified as:

   ```text
   number of texts × 1024
   ```

Although a Spark task defaults to 10 chunks, the ONNX internal batch defaults to 1. That means a worker normally performs ten sequential ONNX calls for a ten-chunk Spark task. Increasing `--embedding-batch-size` changes that internal grouping.

## 10. Returning the result to the driver

The worker creates an ordinary serializable result:

```python

            "chunk_index": 4,
            "text": "...",
            "text_hash": "...",
            "embedding": [0.012, -0.034, ...]  # 1024 floats
```

The NumPy arrays are converted into Python lists so Spark can serialize them cleanly.

The worker verifies that the result does not exceed `--result-bytes`, default 1 MiB. The result then travels:

```text
Python worker
  → executor JVM
  → Spark transport
  → driver JVM/Python

```

`collect()` is the exact point where the embedding result returns to the driver.

It does not write to PostgreSQL from the executor.

## 11. Driver-side validation and database commit

As soon as one Spark future completes, the driver calls:

```python
embedding_store.commit_batch(conn, cfg, result["rows"])
```

That is wired in [embeddings/pipeline.py](/home/spark-master/rag_news_project/spark_warc_v2/embeddings/pipeline.py:43).

Before insertion, [db/embeddings.py](/home/spark-master/rag_news_project/spark_warc_v2/db/embeddings.py:42) checks:

- No duplicate chunk identities exist in the result.
- Returned text still matches its hash.
- Every vector has exactly 1024 numeric values.
- Every value is finite.
- Every vector is L2-normalized.
- The source chunk still exists.
- The source chunk is still eligible.
- Its database hash has not changed.

The driver then performs a bulk insert:

```sql
INSERT INTO chunk_embeddings (...)
VALUES (...)
ON CONFLICT DO NOTHING
```

The transaction covers one completed Spark task batch.

After insertion, it reads the rows back by identity and verifies the stored text hashes. If any validation fails, the transaction rolls back.

The table’s primary identity is:

```text
page_id
+ chunking_version
+ chunk_index
+ embedding_model
+ embedding_version
```

The table is defined in [db/vector_schema.py](/home/spark-master/rag_news_project/spark_warc_v2/db/vector_schema.py:3), with a `VECTOR(1024)` column and an HNSW inner-product index.

Because vectors are normalized, inner product corresponds to cosine-style similarity ranking.

## 12. Progress, failure, and resume behavior

After every committed batch, the driver prints JSON such as:

```json
{"committed": 10}
```

At the end, it queries PostgreSQL and prints something like:

```json
{
  "eligible": 10000,
  "embedded": 9500,
  "pending": 500,
  "embedding_model": "BAAI/bge-m3-int8",
  "embedding_version": "bge-m3-int8-..."
}
```

If one task fails:

- Other already-running jobs in the same small wave are allowed to finish.
- Their successful results are committed.
- The overall application then fails.
- Uncommitted chunks remain pending.

Restarting the same command resumes naturally because the source query excludes embeddings already stored for the same model and version.

Database insertion is idempotent because of the composite primary key and `ON CONFLICT DO NOTHING`, followed by hash verification.

## 2. Why the current code waits for the entire wave

Suppose there are two parallel task slots:

```text
Wave 1:
  Task A → Executor 1
  Task B → Executor 2
```

Imagine A finishes in 10 seconds but B takes 30 seconds.

The current behavior is:

```text
Time 0:  A starts, B starts
Time 10: A finishes and is committed
Time 10–30: Executor 1 may remain unused
Time 30: B finishes and is committed
Time 30: Next wave starts
```

The driver does commit A as soon as it finishes. It does not wait to commit A.

But it does wait for B before submitting the next wave.

That comes from this structure:

```python
wave = list(islice(tasks, parallelism))
futures = [submit(task) for task in wave]

for future in as_completed(futures):
    commit(future.result())

# Only after every future in the wave completes:
# return to the loop and construct the next wave.
```

The implementation is in [jobs/batching.py](/home/spark-master/rag_news_project/spark_warc_v2/jobs/batching.py:27).

### Why it may have been designed this way

The wave boundary provides simple behavior:

- At most `parallelism` tasks are active.
- Failure handling is predictable.
- If one task fails, the code drains and commits successful siblings.
- It does not continue loading and scheduling more database rows after detecting a failure.
- Interrupt handling only needs to drain a small, known set of tasks.
- Driver memory is clearly bounded.

It is a conservative coordination design.

### The downside

It can underutilize executors when task durations differ.

A rolling scheduler would behave like:

```text
Time 0:  A starts, B starts
Time 10: A finishes; C immediately starts
Time 20: C finishes; D immediately starts
Time 30: B finishes; E immediately starts
```

That would keep all executor slots occupied.

A rolling scheduler could still enforce:

```text
active task count <= parallelism
```

but would require slightly more involved handling for:

- What to do after the first failure
- Whether to keep scheduling after a failure has been observed
- How many already-read database batches may be in flight
- Interrupt draining
- Preserving the intended stop semantics

For embeddings, rolling refill would likely improve utilization if chunk batches have noticeably different execution times.

## 3. Why this Spark approach is unusual

The normal Spark pattern is to create one RDD or DataFrame containing all work:

```python
results = (
    spark.sparkContext
    .parallelize(all_batches, number_of_partitions)
    .map(process_batch)
    .collect()
)
```

Spark then receives one job with many partitions:

```text
One Spark job
├── Partition 1 → executor
├── Partition 2 → executor
├── Partition 3 → executor
├── Partition 4 → executor
└── ...
```

Spark’s scheduler automatically assigns the next partition to whichever executor becomes free.

This project instead does:

```python
for each batch:
    parallelize([batch], 1).map(process_batch).collect()
```

and uses driver threads to run several of those calls concurrently.

That produces:

```text
Spark job 1 → one partition → one batch
Spark job 2 → one partition → one batch
Spark job 3 → one partition → one batch
```

So the driver is implementing part of the task coordination that Spark would normally handle.

### Advantages of this project’s approach

Each batch is an independent unit:

```text
submit batch → receive batch result → commit batch
```

That gives the driver:

- Per-batch PostgreSQL transactions
- Immediate commit when a batch finishes
- A small number of results held in driver memory
- Straightforward restart behavior
- Explicit control over how many database-derived batches are active
- The ability to preserve successful sibling results if another batch fails

### Disadvantages

It also introduces:

- One Spark job per batch
- More scheduling overhead
- More driver-side coordination
- The wave underutilization discussed above
- Less opportunity for Spark itself to optimize scheduling
- Many small `parallelize → map → collect` operations

So “unusual” does not mean incorrect. It means the project treats Spark more like a distributed function-execution system than a conventional large RDD/DataFrame transformation engine.

            vectors = self.embedder([item.query for item in queries])
            if np.asarray(vectors).shape != (len(queries), 1024):
                raise ValueError("query embedder must return one 1024-dimensional vector per query")
            with ThreadPoolExecutor(max_workers=self.config.max_concurrent_searches,
                                    thread_name_prefix="rag-search") as executor:
                futures = [
                    executor.submit(self._search_one, query, np.asarray(vector, dtype=np.float32))
                    for query, vector in zip(queries, vectors)
                ]
                return [future.result() for future in futures]
    
        def _search_one(self, query: SearchQuery, vector: np.ndarray) -> list[SearchHit]:
            conn = get_connection()
            try:
                register_vector(conn)
                with conn.cursor() as cur:
                    cur.execute(_SEARCH_SQL, (
                        vector,
                        self.config.chunking_version,
                        self.config.embedding_model,
                        self.config.embedding_version,
                        vector,
                        self.config.top_k,
                    ))
                    rows = cur.fetchall()
                conn.rollback()
            finally:
                conn.close()
    
            hits = []
            for row in rows:
                (page_id, chunk_index, text_hash, text, similarity, url, title,
                 publisher, author, language, published_date) = row
                hits.append(SearchHit(
                    page_id=page_id,
                    chunk_index=chunk_index,
                    text_hash=text_hash,
                    text=text,
                    similarity=float(similarity),
                    query_id=query.id,
                    source=Source(
                        page_id=page_id,
                        url=url or "",
                        title=title or "N/A",
                        publisher=publisher or "Unknown",
                        author=author or publisher or "Unknown",
                        language=language or "",
                        published_date=published_date or "N/A",
                    ),
                ))
            return hits

    def _unique_queries(queries: list[SearchQuery],
                        previous: list[SearchQuery] | None = None) -> list[SearchQuery]:
        seen = {item.query.casefold() for item in (previous or [])}
        result = []
        for item in queries:
            key = item.query.casefold()
            if key not in seen:
                seen.add(key)
                result.append(item)
            if len(result) == 3:
                break
        return result
    
    
    def _answer_language(question: str) -> str:
        if re.search(r"[\u0600-\u06ff]", question):
            return "Arabic; match the user's dialect when possible"
        return "the same language as the original question"
    
    
    def _format_evidence(evidence: list[Evidence]) -> str:
        if not evidence:
            return "(no matching evidence)"
        blocks = []
        for item in evidence:
            source_text = "; ".join(
                f"{source.title} | {source.publisher} | {source.published_date} | {source.url}"
                for source in item.sources
            )
            blocks.append(
                f"Chunk IDs: {', '.join(item.chunk_ids)}\n"
                f"Matched queries: {', '.join(item.query_ids)}\n"
                f"Cosine similarity: {item.similarity:.6f}\n"
                f"Sources: {source_text}\n"
                f"Text: {item.text}"
            )
        return "\n\n---\n\n".join(blocks)
    
    
    class RagAgent:
        """Two-round agent: plan, retrieve, answer or make one corrective search."""
    
        def __init__(self, retriever=None, planner=None, decider=None, finalizer=None):
            self.retriever = retriever or VectorRetriever()
            injected_decider = decider is not None
            if planner is None or decider is None:
                api_key = os.environ.get("GROQ_API_KEY")
                if not api_key:
                    raise ValueError("GROQ_API_KEY is required")
                llm = ChatGroq(
                    model=os.environ.get("RAG_MODEL", "qwen/qwen3.8-27b"),
                    temperature=0,
                    max_tokens=2400,
                    max_retries=2,
                    api_key=api_key,
                    reasoning_effort="none",
                )
                planner = planner or (
                    _PLAN_PROMPT | llm.with_structured_output(QueryPlan, method="json_schema")
                )
                decider = decider or (
                    _DECISION_PROMPT | llm.with_structured_output(AgentDecision, method="json_schema")
                )
                finalizer = finalizer or (
                    _FINAL_ANSWER_PROMPT
                    | llm.with_structured_output(GroundedAnswer, method="json_schema")
                )
            elif finalizer is None and injected_decider:
                # Test/custom chains can handle both decision shapes themselves.
                finalizer = decider
            self.planner = planner
            self.decider = decider
            self.finalizer = finalizer
    
        async def answer(self, question: str) -> RagResult:
            question = question.strip()
            if not question:
                raise ValueError("question must not be empty")
    
            plan = await self.planner.ainvoke({"question": question})
            queries = _unique_queries(plan.queries)
            if not queries:
                raise ValueError("query planner returned no usable queries")
    
            all_hits = await self.retriever.retrieve(queries)
            evidence = assemble_evidence(all_hits)
            rounds = 1
            decision = await self._decide(question, evidence, retry_allowed=True)
    
            if decision.action == "search_more":
                follow_up = _unique_queries(decision.queries, queries)
                if follow_up:
                    queries.extend(follow_up)
                    all_hits.extend(await self.retriever.retrieve(follow_up))
                    evidence = assemble_evidence(all_hits)
                    rounds = 2
                    decision = await self._finalize(question, evidence)
            elif decision.action == "insufficient" and evidence:
                decision = await self._finalize(question, evidence)
    
            return self._result(question, queries, evidence, rounds, decision)
    
        async def _decide(self, question: str, evidence: list[Evidence], retry_allowed: bool):
            return await self.decider.ainvoke({
                "question": question,
                "answer_language": _answer_language(question),
                "retry_allowed": "yes" if retry_allowed else "no",
                "evidence": _format_evidence(evidence),
            })
    
        async def _finalize(self, question: str, evidence: list[Evidence]) -> AgentDecision:
            final = await self.finalizer.ainvoke({
                "question": question,
                "answer_language": _answer_language(question),
                "evidence": _format_evidence(evidence),
            })
            if isinstance(final, AgentDecision):
                return final
            return AgentDecision(
                action="answer" if final.answer else "insufficient",
                answer=final.answer,
                citations=final.citations,
            )
    
        @staticmethod
        def _result(question, queries, evidence, rounds, decision) -> RagResult:
            available = {chunk_id for item in evidence for chunk_id in item.chunk_ids}
            inline_citations = re.findall(r"p[0-9]+:c[0-9]+", decision.answer)
            valid_citations = list(dict.fromkeys(
                citation for citation in [*decision.citations, *inline_citations]
                if citation in available
            ))
            sufficient = decision.action == "answer" and bool(decision.answer) and bool(valid_citations)
            if sufficient:
                answer = decision.answer
            else:
                reason = decision.reason or "The retrieved archive evidence does not support a reliable answer."
                answer = f"Insufficient evidence: {reason}"
    
            cited_pages = {int(item.split(":", 1)[0][1:]) for item in valid_citations}
            sources = []
            seen_sources = set()
            for item in evidence:
                if not cited_pages.intersection(source.page_id for source in item.sources):
                    continue
                for source in item.sources:
                    key = (source.page_id, source.url)
                    if source.page_id in cited_pages and key not in seen_sources:
                        seen_sources.add(key)
                        sources.append(source)
            return RagResult(
                question=question,
                answer=answer,
                sufficient=sufficient,
                citations=valid_citations,
                sources=sources,
                queries=queries,
                evidence=evidence,
                retrieval_rounds=rounds,
            )
    
    
    async def answer(question: str) -> RagResult:
        """Answer one question with the default environment-configured agent."""
        return await RagAgent().answer(question)

    """Deterministic evidence deduplication and adjacent-hit assembly."""
    
    from __future__ import annotations
    
    import hashlib
    
    from retrieval.models import Evidence
    from retrieval.store import SearchHit
    
    
    def _unique(values):
        return list(dict.fromkeys(values))
    
    
    def _merge_overlap(left: str, right: str) -> str:
        """Join chunks while removing their largest exact suffix/prefix overlap."""
        limit = min(len(left), len(right))
        overlap = 0
        for size in range(limit, 0, -1):
            if left[-size:] == right[:size]:
                overlap = size
                break
        separator = "" if overlap or not left or not right else "\n"
        return left + separator + right[overlap:]
    
    
    def assemble_evidence(hits: list[SearchHit]) -> list[Evidence]:
        """Deduplicate hits, merge adjacent hits, then deduplicate exact blocks."""
        physical = {}
        for hit in hits:
            key = (hit.page_id, hit.chunk_index)
            if key not in physical:
                physical[key] = {
                    "hit": hit,
                    "query_ids": [hit.query_id],
                    "similarity": hit.similarity,
                }
            else:
                item = physical[key]
                item["query_ids"] = _unique(item["query_ids"] + [hit.query_id])
                item["similarity"] = max(item["similarity"], hit.similarity)
    
        runs = []
        current = None
        for (page_id, chunk_index), item in sorted(physical.items()):
            hit = item["hit"]
            if current is None or current["page_id"] != page_id or chunk_index != current["last_index"] + 1:
                if current is not None:
                    runs.append(current)
                current = {
                    "page_id": page_id,
                    "last_index": chunk_index,
                    "text": hit.text,
                    "chunk_ids": [hit.chunk_id],
                    "query_ids": list(item["query_ids"]),
                    "similarity": item["similarity"],
                    "sources": [hit.source],
                }
            else:
                current["last_index"] = chunk_index
                current["text"] = _merge_overlap(current["text"], hit.text)
                current["chunk_ids"].append(hit.chunk_id)
                current["query_ids"] = _unique(current["query_ids"] + item["query_ids"])
                current["similarity"] = max(current["similarity"], item["similarity"])
        if current is not None:
            runs.append(current)
    
        by_text = {}
        for run in runs:
            digest = hashlib.sha256(run["text"].encode()).hexdigest()
            if digest not in by_text:
                by_text[digest] = run
                continue
            existing = by_text[digest]
            existing["chunk_ids"] = _unique(existing["chunk_ids"] + run["chunk_ids"])
            existing["query_ids"] = _unique(existing["query_ids"] + run["query_ids"])
            existing["similarity"] = max(existing["similarity"], run["similarity"])
            source_keys = {(source.page_id, source.url) for source in existing["sources"]}
            existing["sources"].extend(
                source for source in run["sources"]
                if (source.page_id, source.url) not in source_keys
            )
    
        evidence = [Evidence(
            text=item["text"],
            chunk_ids=item["chunk_ids"],
            query_ids=item["query_ids"],
            similarity=item["similarity"],
            sources=item["sources"],
        ) for item in by_text.values()]
        return sorted(evidence, key=lambda item: (-item.similarity, item.chunk_ids[0]))

    # Retrieval and Chatbot Design
    
    This document describes the bounded retrieval and answering workflow implemented
    in `retrieval/` and exposed by `scripts/ask.py`.
    
    ## Decisions
    
    - Retrieval is vector-only. Do not add keyword search, BM25, PostgreSQL full-text
      search, `LIKE`, `ILIKE`, `tsvector`, or keyword-derived score adjustments.
    - The retrieval service returns ranked evidence and provenance. The separate
      chatbot module generates answers.
    - Do not put an LLM intent router in front of every search.
    - Do not implement rule-based query decomposition. It becomes brittle across
      Arabic, English, and French and attempts to decide what evidence is needed before
      seeing the first retrieval result.
    - The chatbot may produce one to three self-contained semantic search queries. It
      can decompose a multi-part question when useful.
    - The chatbot may request one additional retrieval round after identifying the
      specific evidence that is missing.
    - Use LangChain Core prompt pipelines plus ordinary Python and `asyncio` for the
      bounded workflow. LangGraph is deferred until durable checkpoints, human
      approval, multiple tools, or more complicated loops justify it.
    - Do not import the sibling fake retriever or its current retry loop. Repeating the
      same deterministic search without changing the query or search parameters is not
      corrective retrieval.
    
    ## Component responsibilities
    
    ### Retrieval service
    
    The retrieval service is deterministic and does not decide user intent. It:
    
    1. Accepts one to three search queries.
    2. Embeds all queries in one batch with the existing BGE-M3 implementation.
    3. Searches the versioned `chunk_embeddings` rows through exact pgvector cosine
       distance.
    4. Returns the top results for each query, preserving the query-to-result mapping.
    5. Deduplicates repeated chunks while retaining every query that matched them.
    6. Returns chunk text, similarity, stable chunk identity, URL, title, language,
       publisher, author, and publication date.
    
    Every search must use the same model files, maximum token length, pooling,
    normalization, `embedding_model`, and computed `embedding_version` used to create
    the stored chunk vectors. Searches must also select the configured
    `chunking_version` and eligible chunks only.
    
    ### Chatbot module
    
    The chatbot owns semantic planning and answering. It:
    
    1. Keeps the original user question unchanged in its state.
    2. Produces one query for a simple question or up to three self-contained queries
       when separate evidence is required.
    3. Examines the grouped evidence and either answers or names the missing aspects.
    4. May request one targeted follow-up search round.
    5. Generates an answer using returned evidence and cites stable chunk identifiers.
    6. Reports insufficient evidence when the allowed retry still does not provide the
       required support.
    
    The chatbot must not answer from an unsupported draft and then use its own prose as
    the retrieval query. Follow-up queries must describe evidence missing from the
    original question.
    
    ## Request flow
    
    ```mermaid
    flowchart TD
        U[User question] --> P[Chatbot creates 1-3 search queries]
        P --> E[BGE-M3 embeds the query batch]
        E --> V[Parallel pgvector searches]
        V --> G[Grouped and deduplicated evidence]
        G --> A{Chatbot evidence assessment}
        A -->|Sufficient| R[Answer with citations]
        A -->|Specific evidence missing| Q[Create targeted follow-up queries]
        Q --> E2[BGE-M3 embeds follow-up batch]
        E2 --> V2[Parallel pgvector searches]
        V2 --> G2[Merge evidence]
        G2 --> F{Final assessment}
        F -->|Sufficient| R
        F -->|Still insufficient| S[Return insufficient evidence]
    ```
    
    Keep results grouped by search query until the chatbot has checked coverage. A
    single global ranking can allow an easy sub-question to consume every result and
    leave another part without evidence.
    
    ## Model responsibilities
    
    | Task | Mechanism |
    |---|---|
    | Query and chunk embeddings | Local BGE-M3 bi-encoder |
    | Corpus candidate ranking | pgvector inner-product similarity |
    | Query decomposition | Chatbot LLM, only when it chooses multiple searches |
    | Missing-evidence analysis | Chatbot LLM after seeing retrieved chunks |
    | Follow-up query generation | Chatbot LLM, at most one round |
    | Answer generation | Separate chatbot LLM |
    | Candidate reranking | Optional multilingual cross-encoder, added only after evaluation |
    | Answer grounding | Optional multilingual NLI cross-encoder in the chatbot module |
    
    A bi-encoder embeds queries and chunks independently, which makes corpus search
    efficient. A reranking cross-encoder reads a query and candidate chunk together and
    produces a more precise relevance score, but it requires one inference per pair.
    If introduced, apply it only to a small pgvector candidate set, such as the top
    10-30 results.
    
    A grounding cross-encoder performs a different task. It reads an answer claim and
    its cited evidence together and predicts entailment, neutrality, or contradiction.
    It can determine whether the evidence supports the claim; it cannot establish that
    the source itself is factually true. Reranking and grounding models are deferred
    until labeled evaluations demonstrate a need.
    
    ## Structured agent contract
    
    The chatbot's search decisions are Pydantic contracts produced through LangChain
    structured output. Application code validates the query and round limits before
    calling retrieval.
    
    ```json
    {
      "action": "search",
      "queries": [
        {
          "id": "comparison",
          "query": "differences between policy A and policy B",
          "purpose": "Find evidence comparing the policies"
        },
        {
          "id": "replacement_reason",
          "query": "why policy B replaced policy A",
          "purpose": "Find the reason for the replacement"
        }
      ]
    }
    ```
    
    After retrieval, the chatbot returns either an answer action or a targeted retry:
    
    ```json
    {
      "action": "search_more",
      "reason": "The comparison is supported, but the replacement reason is missing.",
      "missing_aspects": ["Reason policy B replaced policy A"],
      "queries": [
        {
          "id": "replacement_reason_retry",
          "query": "stated reasons policy B replaced policy A",
          "purpose": "Find the missing replacement rationale"
        }
      ]
    }
    ```
    
    These actions are validated Pydantic models. The application invokes retrieval and
    asks the chatbot for the next structured action. The model never receives database
    credentials or executes code.
    
    ## Concurrency
    
    For a batch of decomposed queries:
    
    1. Call `generate_embeddings()` once with the full list. This is more efficient
       than starting one ONNX invocation per query.
    2. Run the synchronous ONNX batch in an owned coordinator worker so it does not
       block callers and command-line shutdown does not depend on asyncio's global
       executor.
    3. Start one pgvector search per query with `asyncio.gather()`.
    4. Because the project uses synchronous `psycopg2`, run each search in that
       executor and open a separate connection for it. Never share a connection or
       cursor between concurrent searches.
    5. Bound database search concurrency with the executor's worker count.
    
    Initial operational limits are:
    
    ```text
    maximum queries per round:       3
    maximum retrieval rounds:        2
    maximum concurrent DB searches:  3
    embedding inference concurrency: 1 batch per process
    ```
    
    The initial implementation uses exhaustive `vector <=> query_vector` ordering.
    The schema's existing `vector_ip_ops` HNSW index remains untouched but is not used
    by this exact cosine query. Switching to approximate indexed retrieval is confined
    to the retrieval store.
    
    These limits must be enforced by application code rather than accepted from model
    output. The service also controls `top_k`, timeouts, database filters, and permitted
    metadata fields.
    
    ## Sibling implementation disposition
    
    The sibling `query-ret-optimization` work is a prototype rather than a production
    retrieval implementation:
    
    - Its query service demonstrates LLM rewrite, expansion, and decomposition, but a
      separate LLM intent router is unnecessary in the agreed design.
    - Its retrieval service returns hard-coded chunks through `FakeRetriever`.
    - Its CRAG evaluator judges the combined context as sufficient or insufficient; it
      does not rank or rerank chunks.
    - Its retry is simulated and does not perform a meaningfully different search.
    
    Do not copy `fake_retriever`, the existing CRAG service loop, the separate query
    HTTP API, generated evaluation results, manual test scripts, or reports. Individual
    prompts and structured-output utilities may be used as references, but the new
    retrieval module should be implemented around the current PostgreSQL schema,
    BGE-M3 embedding code, and pgvector index.
    
    ## Deferred features and evaluation gates
    
    Start with BGE-M3 and pgvector ranking. Before adding a reranker, CRAG evaluator,
    or grounding model, create a multilingual labeled set of real questions and
    expected chunks. Measure at least Recall@k, MRR, per-language results, latency, and
    the rate at which a corrective search adds useful evidence.
    
    Add a reranker only if relevant chunks are retrieved but ordered poorly. Add a

    ### Ask the vector database
    
    Create a dedicated agent environment, configure PostgreSQL and Groq, and point
    to the same local BGE-M3 model files used to create the stored vectors. This
    keeps both `.venv` and the packaged Spark worker environment unchanged:
    
    ```bash

    .venv-rag/bin/python -m pip install -r requirements/rag-agent.txt
    export GROQ_API_KEY=your-key
    .venv-rag/bin/python scripts/ask.py "What happened, and why did it matter?"
    ```
    
    The command always prints the retrieved evidence after the answer, including
    chunk IDs, cosine scores, matched subqueries, sources, and full chunk text. This
    diagnostic evidence is also printed when the agent reports insufficient evidence.
    
    The importable interface is asynchronous:
    
    ```python
    from retrieval import answer
    
    result = await answer("What happened, and why did it matter?")
    print(result.answer)
    ```
    
    The agent uses `qwen/qwen3.8-27b` to create at most three semantic queries,
    embeds them in one BGE-M3 batch, and runs exact cosine searches concurrently.
    Repeated and identical evidence is deduplicated, and adjacent retrieved chunks
    are joined without expanding to unranked neighbors. One targeted follow-up round
    is permitted when evidence for part of the question is missing.
    
    `TOP_K` defaults to `5` per query and `RAG_DB_CONCURRENCY` defaults to `3`.
    `RAG_MODEL`, `CHUNKING_VERSION`, `EMBEDDING_MODEL`, and `EMBEDDING_VERSION` can
    override their detected defaults. Exact cosine search intentionally does not use
    the current inner-product HNSW index; a cosine or normalized inner-product HNSW
    retriever can replace the database search later without changing the agent.
    
    `RAG_FETCH_BATCH_SIZE` defaults to 1,000 rows. The eager
    `load_documents_distributed()` helper requires `limit=` or `RAG_DOCUMENT_LIMIT`.

    def _unique(values):
        return list(dict.fromkeys(values))
    
    
    def _merge_overlap(left: str, right: str) -> str:
        """Join chunks while removing their largest exact suffix/prefix overlap."""
        limit = min(len(left), len(right))
        overlap = 0
        for size in range(limit, 0, -1):
            if left[-size:] == right[:size]:
                overlap = size
                break
        separator = "" if overlap or not left or not right else "\n"
        return left + separator + right[overlap:]
    
    
    def assemble_evidence(hits: list[SearchHit]) -> list[Evidence]:
        """Deduplicate hits, merge adjacent hits, then deduplicate exact blocks."""
        physical = {}
        for hit in hits:
            key = (hit.page_id, hit.chunk_index)
            if key not in physical:
                physical[key] = {
                    "hit": hit,
                    "query_ids": [hit.query_id],
                    "similarity": hit.similarity,
                }
            else:
                item = physical[key]
                item["query_ids"] = _unique(item["query_ids"] + [hit.query_id])
                item["similarity"] = max(item["similarity"], hit.similarity)
    
        runs = []
        current = None
        for (page_id, chunk_index), item in sorted(physical.items()):
            hit = item["hit"]
            if current is None or current["page_id"] != page_id or chunk_index != current["last_index"] + 1:
                if current is not None:
                    runs.append(current)
                current = {
                    "page_id": page_id,
                    "last_index": chunk_index,
                    "text": hit.text,
                    "chunk_ids": [hit.chunk_id],
                    "query_ids": list(item["query_ids"]),
                    "similarity": item["similarity"],
                    "sources": [hit.source],
                }
            else:
                current["last_index"] = chunk_index
                current["text"] = _merge_overlap(current["text"], hit.text)
                current["chunk_ids"].append(hit.chunk_id)
                current["query_ids"] = _unique(current["query_ids"] + item["query_ids"])
                current["similarity"] = max(current["similarity"], item["similarity"])
        if current is not None:
            runs.append(current)
    
        by_text = {}
        for run in runs:
            digest = hashlib.sha256(run["text"].encode()).hexdigest()
            if digest not in by_text:
                by_text[digest] = run
                continue
            existing = by_text[digest]
            existing["chunk_ids"] = _unique(existing["chunk_ids"] + run["chunk_ids"])
            existing["query_ids"] = _unique(existing["query_ids"] + run["query_ids"])
            existing["similarity"] = max(existing["similarity"], run["similarity"])
            source_keys = {(source.page_id, source.url) for source in existing["sources"]}
            existing["sources"].extend(
                source for source in run["sources"]
                if (source.page_id, source.url) not in source_keys
            )
    
        evidence = [Evidence(
            text=item["text"],
            chunk_ids=item["chunk_ids"],
            query_ids=item["query_ids"],
            similarity=item["similarity"],
            sources=item["sources"],
        ) for item in by_text.values()]
        return sorted(evidence, key=lambda item: (-item.similarity, item.chunk_ids[0]))

    from typing import Iterable, Iterator, List
    from langchain_core.documents import Document
    from langchain_text_splitters import RecursiveCharacterTextSplitter
    from chunking import CHUNKING_VERSION
    
    
    def iter_document_chunks(documents: Iterable[Document], chunk_size: int = 800,
                             chunk_overlap: int = 100) -> Iterator[Document]:
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=chunk_size, chunk_overlap=chunk_overlap,
            separators=["\n\n", "\n", ".", " ", ""],
        )
        for document in documents:
            for index, chunk in enumerate(splitter.split_documents([document])):
                chunk.metadata = dict(chunk.metadata, chunk_index=index)
                yield chunk
    
    
    def split_documents_into_chunks(documents: List[Document], chunk_size: int = 800,
                                    chunk_overlap: int = 100) -> List[Document]:
        return list(iter_document_chunks(documents, chunk_size, chunk_overlap))

    def process_batch(task):
        from langchain_core.documents import Document
        from chunking.preprocessing import clean_article_text, noise_reason
        from chunking.splitters import iter_document_chunks
        result = {"documents": []}
        for source in task["items"]:
            document = Document(page_content=clean_article_text(source["text"]), metadata={
                "doc_id": source["page_id"], "source": source["source"],
                "title": source["title"], "language": source["language"],
            })
            rows = []
            for chunk in iter_document_chunks([document]):
                reason = noise_reason(chunk)
                rows.append({"chunk_index": chunk.metadata["chunk_index"], "chunk_text": chunk.page_content,
                             "text_hash": hashlib.sha256(chunk.page_content.encode()).hexdigest(),
                             "eligible": reason is None, "filter_reason": reason})
            result["documents"].append({"page_id": source["page_id"], "source_hash": chunks.source_hash(source), "chunks": rows})
            if serialized_size(result) > task["result_bytes"]:
                raise ValueError("chunk result exceeds byte limit; reduce --items-per-task or increase --result-bytes")
        return result

# Add BGE-M3 Cross-Encoder Reranking

## Summary

Add `BAAI/bge-reranker-v2-m3` as an online second-stage reranker:

```text
User/search query
    ↓
BGE-M3 query embedding
    ↓
PostgreSQL retrieves top 20 by cosine similarity
    ↓
Cross-encoder scores the 20 (query, chunk) pairs
    ↓
Keep best 5
    ↓
Evidence assembly and answer generation
```

The cross-encoder belongs in the `.venv-rag` query process, not the Spark embedding pipeline. Its score depends on a query and passage together, so it cannot be calculated permanently when corpus embeddings are created. BAAI documents the model as a multilingual reranker that returns one relevance logit per query–passage pair; applying sigmoid converts that logit to a 0–1 score without changing its order. [BAAI model card](https://huggingface.co/BAAI/bge-reranker-v2-m3)

Use the pinned INT8 ONNX conversion at commit `90213ffc6a8e6f051a6331269a0f5526cdd896f6`. Its model file is approximately 571 MB with SHA-256 `912fc1215c2dbff6499700534bd8d31253af01573861abbfc43afd1fab6cce5d`. [Pinned ONNX conversion](https://huggingface.co/onnx-community/bge-reranker-v2-m3-ONNX/commit/90213ffc6a8e6f051a6331269a0f5526cdd896f6)

Behavior:

- Load the tokenizer with `local_files_only=True`.
- Load `onnx/model_int8.onnx` through `CPUExecutionProvider`.

### `sigmoid()`

```python
def sigmoid(logits: np.ndarray) -> np.ndarray:
    ...
```

Purpose:

> Convert raw model logits into understandable scores between zero and one.

Mathematically:

```text
sigmoid(x) = 1 / (1 + exp(-x))
```

Implement it in a numerically stable way because directly calculating `exp(1000)` can overflow.

Expected examples:

```text

logit   0 → score 0.5
logit  10 → score close to 1
```

The sigmoid does not change result ordering. A larger logit always produces a larger normalized score.

### `OnnxReranker.ensure_loaded()`

```python
    self,
    pairs: list[tuple[str, str]],

    ...
```

Purpose:

> Convert query–passage pairs into the NumPy tensors expected by ONNX.

Split the pairs into parallel lists:

```python
queries = [query for query, passage in pairs]
```

### `OnnxReranker._run_batch()`

```python

    self,
    pairs: list[tuple[str, str]],
) -> np.ndarray:
    ...
```

Purpose:

> Run one small batch through ONNX and return raw validated logits.

It should:

1. Reject an empty batch or return an empty array consistently.
2. Call `ensure_loaded()`.
3. Call `_tokenize(pairs)`.
4. Acquire `_inference_lock`.
5. Call `self._session.run(...)`.
6. Pass the model output to `validate_logits()`.
7. Return raw logits, not sigmoid scores.

Keeping sigmoid outside this function makes `_run_batch()` correspond directly to one model execution.

### `OnnxReranker.score()`

```python
def score(
    self,
    pairs: list[tuple[str, str]],
) -> np.ndarray:
    ...
```

Purpose:

> Provide the public interface for scoring any number of pairs.

It should:

1. Return an empty one-dimensional array for no pairs.
2. Validate that every query and passage is a string.
3. Divide the pairs into groups of `config.batch_size`.
4. Call `_run_batch()` for each group.
5. Concatenate the raw logits in original input order.
6. Validate the combined count.
7. Apply `sigmoid()`.
8. Return one normalized score for every input pair.

Pseudocode:

```python
def score(self, pairs):
    if not pairs:
        return np.empty(0, dtype=np.float32)

    batches = []
    for start in range(0, len(pairs), self.config.batch_size):
        batch = pairs[start:start + self.config.batch_size]
        batches.append(self._run_batch(batch))

    logits = np.concatenate(batches)
    logits = validate_logits(logits, len(pairs))
    return sigmoid(logits)
```

`score()` is an orchestrator, but it remains short because tokenization, inference, validation, and normalization are separate functions.
